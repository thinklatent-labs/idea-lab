from relay import Client

client = Client(use_case="UC-0307")                 # Fund document Q&A
run = client.agents.run(
    agent="knowledge-agent:prod",                   # alias; "knowledge-agent:2.3" pins a version
    input="Which share class has the lowest expense ratio in the 2025 prospectus?",
    on_behalf_of=user.id,                           # answers use only documents this person may open
    tags={"env": "build", "cost_centre": "IM-2210", "app": "fund-qa-web"},   # required on every call
)
print(run.answer, run.citations, run.trace_id)

client.feedback(                                    # counts toward acceptance and corrections
    trace_id=run.trace_id,
    outcome="corrected",                            # accepted, corrected or rejected
    correction="low-cost shares 0.11%, 2025 prospectus p. 14",
)
